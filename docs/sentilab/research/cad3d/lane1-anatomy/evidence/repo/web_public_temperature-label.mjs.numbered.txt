    59	// The rule over given lists (the tests build their own; the pages use the frozen lists above).
    60	export function labeller({ withdrawn = WITHDRAWN, illustrative = ILLUSTRATIVE, accepted = ACCEPTED } = {}) {
    61	  const withdrawnKeys = new Set(withdrawn.map((id) => id.toLowerCase()));
    62	  const illustrativeIds = new Set(illustrative), acceptedIds = new Set(accepted);
    63	  // One value, from one identity.
    64	  const labelFor = (identity) => {
    65	    if (typeof identity !== "string" || identity.trim() === "") return LABELS.unvalidated;   // unvalidated provenance
    66	    if (withdrawnKeys.has(evaluatorKey(identity))) return LABELS.withdrawn;                   // every build of it
    67	    if (illustrativeIds.has(identity)) return LABELS.illustrative;                            // exactly the fixture
    68	    if (acceptedIds.has(identity)) return LABELS.accepted;                                    // exactly as accepted
    69	    return LABELS.unvalidated;
    70	  };
    71	  // A value derived from several: the identities of everything it was made from.
    72	  const labelForAll = (identities) => {
    73	    const list = typeof identities === "string" ? [identities] : Array.isArray(identities) ? identities
    74	      : identities != null && typeof identities[Symbol.iterator] === "function" ? [...identities] : [];
    75	    const labels = list.map(labelFor);
    76	    if (labels.includes(LABELS.withdrawn)) return LABELS.withdrawn;
    77	    if (labels.length > 0 && labels.every((label) => label === LABELS.illustrative)) return LABELS.illustrative;
    78	    if (labels.length > 0 && labels.every((label) => label === LABELS.accepted)) return LABELS.accepted;
    79	    return LABELS.unvalidated;                    // an unvalidated input, a mix of fixture and real inputs, or none
    80	  };
    81	  return Object.freeze({ labelFor, labelForAll });
    82	}

    87	// ── the run view and the verify page: flight records ─────────────────────────────────────────────────────
    88	// A run (a GET /api/runs/{id} response, a replay file of one, or the raw { run_id, lane_results } it wraps), or one
    89	// lane's record document ({ events, ... }, as the TFL-08 records carry it). The identity of every temperature it holds,
    90	// in record order: each baseline, each of its shared initial samples, and each oracle result (a committed proposal's
    91	// temperature is its oracle result's).
    92	export function runTemperatureIds(run) {
    93	  const ids = [];
    94	  const result = isObj(run) && isObj(run.result) ? run.result : run;
    95	  const records = !isObj(result) ? []
    96	    : isObj(result.lane_results) ? Object.values(result.lane_results).map((lane) => (isObj(lane) ? lane.record : null))
    97	    : [result.events];
    98	  const solverOf = (thermal) => (isObj(thermal) ? thermal.solver_version : undefined);
    99	  for (const record of records) {
   100	    for (const e of Array.isArray(record) ? record : []) {
   101	      if (!isObj(e) || (e.event !== "baseline" && e.event !== "oracle_result")) continue;
   102	      const payload = isObj(e.payload) ? e.payload : {};
   103	      ids.push(solverOf(payload.thermal));
   104	      if (e.event === "baseline" && Array.isArray(payload.shared_initial_samples)) {
   105	        for (const sample of payload.shared_initial_samples) ids.push(isObj(sample) ? solverOf(sample.thermal) : undefined);
   106	      }
   107	    }
   108	  }
   109	  return ids;
   110	}

   121	// ── view envelopes (Lab and Package views) ──────────────────────────────────────────────────────────────────
   122	// node id -> evaluator identity, from the node.create the projector APPLIED for each id: the first one it did not
   123	// reject, the same event the views take the node's values from. A rejected create (missing semantics, an unknown kind,
   124	// a repeat) lends no identity. rejected: project()'s rejections ({ event_id, code } entries, or event ids); the pages
   125	// pass their verification's projection.rejected. Only measurements and comparisons declare semantics, so a candidate
   126	// maps to undefined.
   127	export function evaluatorIds(envelopes, rejected = []) {
   128	  const skip = new Set((Array.isArray(rejected) ? rejected : []).map((r) => (isObj(r) ? r.event_id : r)));
   129	  const ids = new Map();
   130	  for (const env of Array.isArray(envelopes) ? envelopes : []) {
   131	    const body = isObj(env) && isObj(env.event) && env.event.type === "node.create" && isObj(env.event.body) ? env.event.body : null;
   132	    if (!body || typeof body.id !== "string" || ids.has(body.id) || skip.has(env.event.event_id)) continue;
   133	    ids.set(body.id, isObj(body.semantics) ? body.semantics.evaluator_version : undefined);
   134	  }
   135	  return ids;
   136	}

   156	export function envelopeEvaluator(env, ids) {
   157	  if (!isObj(env) || !isObj(env.event) || !isObj(env.event.body)) return undefined;
   158	  const body = env.event.body;
