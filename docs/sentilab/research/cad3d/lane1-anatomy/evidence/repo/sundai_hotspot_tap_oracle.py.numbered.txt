   168	def _load_problem(*, config_path: Path = TAP_CONFIGS[TAP_CONFIG_DEFAULT_ID],
   169	                  hotspot_binary: Path | None = HOTSPOT_BINARY) -> TapProblem:
   170	    """Load the TAP problem. hotspot_binary=None gives a geometry-only problem with no solver identity."""
   171	    if not config_path.is_file() or not TAP_HOTSPOT_CONFIG.is_file():
   172	        raise HotSpotOracleError(f"TAP fixture configuration is missing: {config_path}")
   173	    if hotspot_binary is not None and not hotspot_binary.is_file():
   174	        raise HotSpotOracleError(f"HotSpot binary is not built: {hotspot_binary}")
   175	
   176	    parser = configparser.ConfigParser(interpolation=None)
   177	    parser.read_string(_read_text(config_path))
   178	    count = parser.getint("chiplets", "chiplet_count")
   179	    widths = _parse_float_list(parser.get("chiplets", "widths"), "widths")
   180	    heights = _parse_float_list(parser.get("chiplets", "heights"), "heights")
   181	    powers = _parse_float_list(parser.get("chiplets", "powers"), "powers")
   182	    rows = parser.get("chiplets", "connections").split(";")
   183	    try:
   184	        connections = tuple(tuple(float(value.strip()) for value in row.split(",") if value.strip())
   185	                            for row in rows if row.strip())
   186	        connections = tuple(tuple(float(value) for value in row) for row in connections)
   187	        size = parser.getfloat("interposer", "intp_size")
   188	        granularity = parser.getfloat("general", "placer_granularity", fallback=1.0)
   189	        link_type = parser.get("interposer", "link_type", fallback="nppl")
   190	    except (ValueError, configparser.Error) as exc:
   191	        raise HotSpotOracleError("invalid TAP fixture configuration") from exc
   192	    if (count != len(widths) or count != len(heights) or count != len(powers) or
   193	            len(connections) != count or any(len(row) != count for row in connections) or
   194	            not math.isfinite(size) or size <= 0 or not math.isfinite(granularity) or granularity <= 0):
   195	        raise HotSpotOracleError("inconsistent TAP fixture configuration")
   196	
   197	    ambient_kelvin = None
   198	    for line in _read_text(TAP_HOTSPOT_CONFIG).splitlines():
   199	        parts = line.split()
   200	        if parts and parts[0] == "-ambient" and len(parts) == 2:
   201	            ambient_kelvin = float(parts[1])
   202	            break
   203	    if ambient_kelvin is None or not math.isfinite(ambient_kelvin) or ambient_kelvin <= 0:
   204	        raise HotSpotOracleError("TAP HotSpot config has no valid ambient temperature")
   205	
   206	    system = PassiveInterposer()
   207	    system.set_chiplet_count(count)
   208	    system.initialize()
   209	    system.set_link_type(link_type)
   210	    system.set_chiplet_size(list(widths), list(heights))
   211	    system.set_connection_matrix([list(row) for row in connections])
   212	    system.compute_ubump_overhead()
   213	    bump_halos = tuple(float(value) for value in system.hubump)
   214	
   215	    generator_paths = (
   216	        config_path,
   217	        TAP_HOTSPOT_CONFIG,
   218	        TAP_ROOT / "UPSTREAM_COMMIT",
   219	        HOTSPOT_ROOT / "UPSTREAM_COMMIT",
   220	        TAP_ROOT / "passive_interposer.py",
   221	        TAP_ROOT / "system.py",
   222	        TAP_ROOT / "init_placement.py",
   223	        TAP_ROOT / "block_occupation.py",
   224	        TAP_ROOT / "fastSA.py",
   225	        TAP_ROOT / "bstree.py",
   226	        TAP_ROOT / "util" / "fill_space.py",
   227	        Path(__file__).resolve(),
   228	        WORKER,
   229	    )
   230	    generators = tuple((path.relative_to(REPO_ROOT).as_posix(), _sha256(path)) for path in generator_paths)
   231	    fixture_id = next(
   232	        (fixture_id for fixture_id, fixture_path in TAP_CONFIGS.items() if fixture_path.resolve() == config_path.resolve()),
   233	        None,
   234	    )
   235	    if fixture_id is None:
   236	        raise HotSpotOracleError(f"unknown TAP fixture for config path: {config_path}")
   237	    return TapProblem(
   238	        fixture_id=fixture_id,
   239	        interposer_size_mm=size,
   240	        widths_mm=widths,
   241	        heights_mm=heights,
   242	        powers_w=powers,
   243	        connections=connections,
   244	        link_type=link_type,
   245	        granularity_mm=granularity,
   246	        bump_halos_mm=bump_halos,
   247	        ambient_c=ambient_kelvin - 273.15,
   248	        config_sha256=_sha256(config_path),
   249	        hotspot_config_sha256=_sha256(TAP_HOTSPOT_CONFIG),
   250	        tap_upstream_commit=_read_upstream_commit(TAP_ROOT / "UPSTREAM_COMMIT"),
   251	        hotspot_upstream_commit=_read_upstream_commit(HOTSPOT_ROOT / "UPSTREAM_COMMIT"),
   252	        hotspot_binary_sha256="" if hotspot_binary is None else _sha256(hotspot_binary),
   253	        generator_sha256=generators,
   254	    )

   257	def _load_legality(config_path: Path, problem: TapProblem) -> TapLegality:
   258	    """Read the optional [sentilab] section; anything unknown or malformed fails closed."""
   259	    parser = configparser.ConfigParser(interpolation=None)
   260	    parser.read_string(_read_text(config_path))
   261	    rules = dict(parser.items(SENTILAB_RULES_SECTION)) if parser.has_section(SENTILAB_RULES_SECTION) else {}
   262	    unknown = set(rules) - {"center_lattice_mm", "min_die_gap_mm", "rotation"}
   263	    if unknown:
   264	        raise HotSpotOracleError(f"unknown SentiLab fixture rule: {sorted(unknown)}")
   265	    try:
   266	        lattice = float(rules.get("center_lattice_mm", problem.granularity_mm))
   267	        min_gap = float(rules.get("min_die_gap_mm", 0.0))
   268	    except ValueError as exc:
   269	        raise HotSpotOracleError("invalid SentiLab fixture rule") from exc
   270	    rotation = rules.get("rotation", "catalog")
   271	    if (not math.isfinite(lattice) or lattice <= 0 or not math.isfinite(min_gap) or min_gap < 0 or
   272	            rotation not in ROTATION_MODES):
   273	        raise HotSpotOracleError("invalid SentiLab fixture rule")
   274	    # A square die has one footprint, so a turn never gives it a second catalog entry.
   275	    catalog = tuple(((w, h),) if rotation == "catalog" or w == h else ((w, h), (h, w))
   276	                    for w, h in zip(problem.widths_mm, problem.heights_mm))
   277	    return TapLegality(TAP_MARGIN_MM, lattice, min_gap, rotation, catalog)

   280	class TapGeometry:
   281	    """A fixture's geometry and legality. Needs no HotSpot binary; HotSpotOracle delegates here."""
   282	
   283	    def __init__(self, problem: TapProblem, legality: TapLegality):
   284	        self.problem = problem
   285	        self.legality = legality
   286	
   287	    @classmethod
   288	    def load(cls, fixture: str = TAP_CONFIG_DEFAULT_ID) -> "TapGeometry":
   289	        if not isinstance(fixture, str) or fixture not in TAP_CONFIGS:
   290	            raise ValueError(f"fixture must be one of {sorted(TAP_CONFIGS)}")
   291	        problem = _load_problem(config_path=TAP_CONFIGS[fixture], hotspot_binary=None)
   292	        return cls(problem, _load_legality(TAP_CONFIGS[fixture], problem))
   293	
   294	    @property
   295	    def fixture_id(self) -> str:
   296	        return self.problem.fixture_id
   297	
   298	    def design_from_centers(self, centers_mm, rotated=None) -> Design:
   299	        """Build the controller Design for this fixture. rotated[i] turns die i by 90 degrees."""
   300	        count = len(self.problem.widths_mm)
   301	        if len(centers_mm) != count:
   302	            raise TapLegalityError("TAP layout must provide one center per chiplet")
   303	        rotated = (False,) * count if rotated is None else tuple(rotated)
   304	        if len(rotated) != count or any(not isinstance(flag, bool) for flag in rotated):
   305	            raise TapLegalityError("TAP layout must give one boolean rotation flag per chiplet")
   306	        chiplets, positions = [], []
   307	        for index, ((x, y), width, height, turned) in enumerate(
   308	                zip(centers_mm, self.problem.widths_mm, self.problem.heights_mm, rotated)):
   309	            if turned:
   310	                width, height = height, width
   311	            chiplets.append(Chiplet(f"tap-chiplet-{index}", width, height))
   312	            positions.append((float(x) - width / 2, float(y) - height / 2))
   313	        return Design(self.problem.interposer_size_mm, self.problem.interposer_size_mm, 0.0,
   314	                      tuple(chiplets), tuple(positions))
   315	
   316	    def rotated(self, candidate: Design) -> tuple[bool, ...]:
   317	        """Which dies of a legal candidate sit turned 90 degrees from the configured orientation."""
   318	        return tuple(not (_same_mm(chiplet.width_mm, width) and _same_mm(chiplet.height_mm, height))
   319	                     for chiplet, width, height in zip(candidate.chiplets, self.problem.widths_mm,
   320	                                                       self.problem.heights_mm))
   321	
   322	    def centers_for(self, candidate: Design) -> tuple[tuple[float, float], ...]:
   323	        """The search gate: every fixture rule, including the centre lattice."""
   324	        return self._checked_centers(candidate, lattice=True)
   325	
   326	    def continuous_centers_for(self, candidate: Design) -> tuple[tuple[float, float], ...]:
   327	        """The continuous geometry rules without the search lattice: frame, orientation catalog,
   328	        margin with halo, halo non-overlap and minimum spacing. For fixed reference layouts such as
   329	        the raw figure reconstructions (gate zero); search candidates go through centers_for."""
   330	        return self._checked_centers(candidate, lattice=False)
   331	
   332	    def _checked_centers(self, candidate: Design, *, lattice: bool) -> tuple[tuple[float, float], ...]:
   333	        if not isinstance(candidate, Design):
   334	            raise TapLegalityError("TAP oracle requires a controller Design")
   335	        try:
   336	            candidate.check()
   337	        except (TypeError, ValueError, OverflowError) as exc:
   338	            raise TapLegalityError(f"invalid TAP candidate: {exc}") from exc
   339	        catalog = self.legality.orientation_catalog_mm
   340	        footprints = tuple((chiplet.width_mm, chiplet.height_mm) for chiplet in candidate.chiplets)
   341	        if (len(footprints) != len(catalog) or
   342	                any(not any(_same_mm(actual[0], width) and _same_mm(actual[1], height) for width, height in allowed)
   343	                    for actual, allowed in zip(footprints, catalog))):
   344	            raise TapLegalityError("TAP fixture chiplet catalog mismatch")
   345	        size = self.problem.interposer_size_mm
   346	        if not _same_mm(candidate.width_mm, size) or not _same_mm(candidate.height_mm, size):
   347	            raise TapLegalityError("TAP fixture interposer size mismatch")
   348	        violations = self.layout_violations(candidate, lattice=lattice)
   349	        if violations:
   350	            raise TapLegalityError(violations[0].message)
   351	        return self._centers(candidate)
   352	
   353	    @staticmethod
   354	    def _centers(candidate: Design) -> tuple[tuple[float, float], ...]:
   355	        # Centres come from each die's own footprint, so a turned die is placed as turned.
   356	        return tuple((float(x + chiplet.width_mm / 2), float(y + chiplet.height_mm / 2))
   357	                     for chiplet, (x, y) in zip(candidate.chiplets, candidate.positions))
   358	
   359	    def layout_violations(self, candidate: Design, *, lattice: bool = True) -> tuple[LayoutViolation, ...]:
   360	        """Every placement-rule miss, in the order centers_for reports them: margins, lattice (unless
   361	        lattice=False), halo overlaps, then minimum spacing. Reads each die's own (possibly turned)
   362	        footprint."""
   363	        size = self.problem.interposer_size_mm
   364	        centers = self._centers(candidate)
   365	        found = []
   366	        margin = self.legality.margin_mm
   367	        rectangles = []
   368	        for index, (chiplet, (center_x, center_y), halo) in enumerate(
   369	                zip(candidate.chiplets, centers, self.problem.bump_halos_mm)):
   370	            rect = (center_x - chiplet.width_mm / 2 - halo,
   371	                    center_y - chiplet.height_mm / 2 - halo,
   372	                    center_x + chiplet.width_mm / 2 + halo,
   373	                    center_y + chiplet.height_mm / 2 + halo)
   374	            excess = max(margin - rect[0], margin - rect[1], rect[2] - (size - margin), rect[3] - (size - margin))
   375	            if excess > 1e-9:
   376	                found.append(LayoutViolation("margin", (index,), excess,
   377	                             f"chiplet {index} violates the {margin:g} mm interposer margin with its microbump halo"))
   378	            rectangles.append(rect)
   379	
   380	        step = self.legality.center_lattice_mm if lattice else None
   381	        if step is not None:
   382	            for index, (center_x, center_y) in enumerate(centers):
   383	                off = max(abs(center_x / step - round(center_x / step)), abs(center_y / step - round(center_y / step)))
   384	                if off > 1e-9:
   385	                    found.append(LayoutViolation("lattice", (index,), off * step,
   386	                                 f"chiplet {index} center is off the {step:g} mm placement grid"))
   387	
   388	        for index, rect in enumerate(rectangles):
   389	            for other_index, other in enumerate(rectangles[:index]):
   390	                if not (rect[2] <= other[0] or other[2] <= rect[0] or
   391	                        rect[3] <= other[1] or other[3] <= rect[1]):
   392	                    depth = min(min(rect[2], other[2]) - max(rect[0], other[0]),
   393	                                min(rect[3], other[3]) - max(rect[1], other[1]))
   394	                    found.append(LayoutViolation("halo_overlap", (other_index, index), depth,
   395	                                 f"chiplets {other_index} and {index} overlap with their microbump halos"))
   396	
   397	        min_gap = self.legality.min_die_gap_mm
   398	        if min_gap > 0:
   399	            # TAP-2.5D Eqn. (10): die bodies at least w_gap apart along some axis.
   400	            bodies = [(rect[0] + halo, rect[1] + halo, rect[2] - halo, rect[3] - halo)
   401	                      for rect, halo in zip(rectangles, self.problem.bump_halos_mm)]
   402	            for index, body in enumerate(bodies):
   403	                for other_index, other in enumerate(bodies[:index]):
   404	                    spacing = max(other[0] - body[2], body[0] - other[2], other[1] - body[3], body[1] - other[3])
   405	                    if spacing < min_gap - 1e-9:
   406	                        apart = (f"are {spacing:.4f} mm apart" if spacing >= 0 else
   407	                                 f"overlap by {-spacing:.4f} mm")
   408	                        found.append(LayoutViolation("min_spacing", (other_index, index), min_gap - spacing,
   409	                                     f"chiplets {other_index} and {index} {apart}, "
   410	                                     f"under the {min_gap:g} mm minimum spacing"))
   411	        return tuple(found)

   413	    def wirelength_mm(self, centers: tuple[tuple[float, float], ...]) -> float:
   414	        total_weight = sum(self.problem.connections[i][j]
   415	                           for i in range(len(centers)) for j in range(i + 1, len(centers)))
   416	        if total_weight <= 0 or not math.isfinite(total_weight):
   417	            raise HotSpotOracleError("TAP benchmark has no positive connection weight")
   418	        distance = sum(self.problem.connections[i][j] *
   419	                       (abs(centers[i][0] - centers[j][0]) + abs(centers[i][1] - centers[j][1]))
   420	                       for i in range(len(centers)) for j in range(i + 1, len(centers)))
   421	        return distance / total_weight

   429	def _number(value: object, label: str) -> float:
   430	    if isinstance(value, bool) or not isinstance(value, numbers.Real):
   431	        raise HotSpotOracleError(f"worker returned invalid {label}")
   432	    result = float(value)
   433	    if not math.isfinite(result):
   434	        raise HotSpotOracleError(f"worker returned non-finite {label}")

   438	async def _terminate_process_group(process: asyncio.subprocess.Process, *, immediate=False) -> None:
   439	    if os.name == "posix":
   440	        if immediate:
   441	            try:
   442	                os.killpg(process.pid, signal.SIGKILL)
   443	            except ProcessLookupError:
   444	                pass
   445	            await process.wait()
   446	            return
   447	        try:
   448	            os.killpg(process.pid, signal.SIGTERM)
   449	        except ProcessLookupError:
   450	            pass
   451	        try:
   452	            await asyncio.wait_for(process.wait(), timeout=1.0)
   453	        except asyncio.TimeoutError:
   454	            pass
   455	        # The worker can exit on SIGTERM while HotSpot or another descendant
   456	        # ignores it and keeps the inherited pipes open. Kill the isolated group
   457	        # after the grace period even when the group leader has already exited.
   458	        try:
   459	            os.killpg(process.pid, signal.SIGKILL)
   460	        except ProcessLookupError:
   461	            pass
   462	        if process.returncode is None:
   463	            await process.wait()
   464	    else:
   465	        taskkill = None
   466	        if process.returncode is None:
   467	            try:
   468	                taskkill = await asyncio.create_subprocess_exec(
   469	                    "taskkill.exe", "/PID", str(process.pid), "/T", "/F",
   470	                    stdout=asyncio.subprocess.DEVNULL,
   471	                    stderr=asyncio.subprocess.DEVNULL,
   472	                    creationflags=subprocess.CREATE_NO_WINDOW,
   473	                )
   474	                await asyncio.wait_for(taskkill.wait(), timeout=3.0)
   475	            except (FileNotFoundError, OSError, asyncio.TimeoutError):
   476	                if taskkill is not None and taskkill.returncode is None:
   477	                    taskkill.kill()
   478	                    await taskkill.wait()
   479	        if process.returncode is None:
   480	            process.kill()
   481	            await process.wait()
   482

   484	async def _uncancellable_cleanup(coroutine) -> None:
   485	    """Finish process cleanup even when the caller is being cancelled."""
   486	    task = asyncio.create_task(coroutine)
   487	    try:
   488	        await asyncio.shield(task)
   489	    except asyncio.CancelledError:
   490	        await task
   491	        raise

   494	class _WarmWorker:
   495	    """One persistent evaluator process and its private parent-owned workspace."""
   496	
   497	    def __init__(self, process: asyncio.subprocess.Process, workspace: tempfile.TemporaryDirectory):
   498	        self.process = process
   499	        self.workspace = workspace
   500	        self.broken = False
   501	        self._retained_directory = None
   502	        self._stderr_file = Path(workspace.name) / 'worker-stderr.bin'
   503	        self._stderr_tail = bytearray()
   504	        self._stderr_task = asyncio.create_task(self._drain_stderr())
   505	        self._stdout_file = Path(workspace.name) / 'worker-response.bin'
   506	        self._response = None
   507	        self._response_buffer = bytearray()
   508	        self._stdout_status = {'retained_bytes': 0, 'has_terminator': False,
   509	                               'eof': False, 'truncated': False, 'parser_error': None}
   510	        self._stdout_task = asyncio.create_task(self._drain_stdout())
   511	
   512	    @classmethod
   513	    async def start(cls) -> "_WarmWorker":
   514	        workspace = tempfile.TemporaryDirectory(prefix="tfl-hotspot-worker-")
   515	        process_options = ({"start_new_session": True} if os.name == "posix" else
   516	                           {"creationflags": subprocess.CREATE_NEW_PROCESS_GROUP})
   517	        try:
   518	            process = await asyncio.create_subprocess_exec(
   519	                sys.executable,
   520	                str(WORKER),
   521	                cwd=str(TAP_ROOT),
   522	                stdin=asyncio.subprocess.PIPE,
   523	                stdout=asyncio.subprocess.PIPE,
   524	                stderr=asyncio.subprocess.PIPE,
   525	                limit=MAX_WORKER_RESPONSE_BYTES,
   526	                env=_worker_environment(),
   527	                **process_options,
   528	            )
   529	        except BaseException:
   530	            workspace.cleanup()
   531	            raise
   532	        return cls(process, workspace)

   534	    @property
   535	    def pid(self) -> int:
   536	        return self.process.pid
   537	
   538	    @property
   539	    def healthy(self) -> bool:
   540	        return not self.broken and self.process.returncode is None
   541	
   542	    async def _drain_stderr(self) -> None:
   543	        assert self.process.stderr is not None
   544	        while True:
   545	            chunk = await self.process.stderr.read(4096)
   546	            if not chunk:
   547	                return
   548	            with self._stderr_file.open('ab') as stream:
   549	                stream.write(chunk)
   550	            self._stderr_tail.extend(chunk)
   551	            if len(self._stderr_tail) > 8192:
   552	                del self._stderr_tail[:-8192]
   553	
   554	    async def _stderr_text(self) -> str:
   555	        if self.process.returncode is not None:
   556	            try:
   557	                await asyncio.wait_for(asyncio.shield(self._stderr_task), timeout=1.0)
   558	            except asyncio.TimeoutError:
   559	                pass
   560	        return self._stderr_tail.decode("utf-8", errors="replace")[-2000:].strip()
   561	
   562	    def _write_stdout_status(self):
   563	        status = dict(self._stdout_status)
   564	        status['status'] = ('terminated' if status['has_terminator'] else 'partial')
   565	        self._stdout_file.with_name('WORKER-RESPONSE.json').write_text(
   566	            json.dumps(status, sort_keys=True) + '\n')
   567	
   568	    async def _drain_stdout(self):
   569	        """Retain bytes independently of parser limits and request cancellation."""
   570	        assert self.process.stdout is not None
   571	        self._stdout_file.touch()
   572	        while True:
   573	            chunk = await self.process.stdout.read(4096)
   574	            if not chunk:
   575	                self._stdout_status['eof'] = True
   576	                self._write_stdout_status()
   577	                if self._response is not None and not self._response.done():
   578	                    self._response.set_result(bytes(self._response_buffer))
   579	                return
   580	            with self._stdout_file.open('ab') as stream:
   581	                stream.write(chunk)
   582	            self._stdout_status['retained_bytes'] += len(chunk)
   583	            self._stdout_status['has_terminator'] |= b'\n' in chunk
   584	            if self._response is not None and not self._response.done():
   585	                prefix, separator, _ = chunk.partition(b'\n')
   586	                if len(self._response_buffer) + len(prefix) + bool(separator) > MAX_WORKER_RESPONSE_BYTES:
   587	                    self._stdout_status['parser_error'] = 'frame_limit_exceeded'
   588	                    self._response.set_exception(HotSpotOracleError('HotSpot worker response exceeds frame limit'))
   589	                    self._response_buffer.clear()
   590	                else:
   591	                    self._response_buffer.extend(prefix + separator)
   592	                    if separator:
   593	                        self._response.set_result(bytes(self._response_buffer))
   594	                        self._response_buffer.clear()
   595	            self._write_stdout_status()
   596	
   597	    async def _exchange(self, payload: bytes) -> bytes:
   598	        assert self.process.stdin is not None
   599	        self._response_buffer.clear()
   600	        self._response = asyncio.get_running_loop().create_future()
   601	        self.process.stdin.write(payload + b"\n")
   602	        await self.process.stdin.drain()
   603	        # A timeout cancels the waiter, never the independent evidence spool.
   604	        return await self._response
   605	
   606	    async def evaluate(self, request: dict, *, worker_timeout_s: float) -> dict:
   607	        if request.get('retained_output_root'):
   608	            self._retained_directory = Path(request['retained_output_root'])
   609	            self._retained_directory.mkdir(parents=True, exist_ok=True)
   610	            destination = self._retained_directory / 'worker-response.bin'
   611	            if self._stdout_file != destination:
   612	                # A cold worker may have emitted startup bytes before admission.
   613	                if self._retained_directory is not None and self._stdout_file.parent == Path(self.workspace.name):
   614	                    if self._stdout_file.exists():
   615	                        shutil.copyfile(self._stdout_file, destination)
   616	                    else:
   617	                        destination.touch()
   618	                else:
   619	                    destination.touch()
   620	                    self._stdout_status = {'retained_bytes': 0, 'has_terminator': False,
   621	                                           'eof': False, 'truncated': False, 'parser_error': None}
   622	                self._stdout_file = destination
   623	                self._write_stdout_status()
   624	            # Spool in the retained tree while the call runs: SIGKILL cannot
   625	            # execute close() to copy a temporary spool afterwards.
   626	            destination = self._retained_directory / 'worker-stderr.bin'
   627	            if self._stderr_file != destination:
   628	                if self._stderr_file.exists():
   629	                    shutil.copyfile(self._stderr_file, destination)
   630	                else:
   631	                    destination.touch()
   632	                self._stderr_file = destination
   633	        self.immediate_kill = request.get('bounded_call', False)
   634	        if not self.healthy:
   635	            self.broken = True
   636	            raise HotSpotOracleError("HotSpot worker is not available")
   637	        message = dict(request)
   638	        message["workspace_root"] = self.workspace.name
   639	        try:
   640	            encoded = json.dumps(message, separators=(",", ":"), allow_nan=False).encode("utf-8")
   641	        except (TypeError, ValueError) as exc:
   642	            self.broken = True
   643	            raise HotSpotOracleError("HotSpot worker request is not valid JSON") from exc
   644	        if len(encoded) > MAX_WORKER_REQUEST_BYTES:
   645	            self.broken = True
   646	            raise HotSpotOracleError("HotSpot worker request exceeds 64 KiB")
   647	        if self._retained_directory is not None:
   648	            (self._retained_directory / 'WORKER.json').write_text(json.dumps({
   649	                'command': [sys.executable, str(WORKER)], 'cwd': str(TAP_ROOT),
   650	                'environment': _worker_environment(), 'request': message}, allow_nan=False) + '\n')
   651	            (self._retained_directory / 'worker-request.bin').write_bytes(encoded + b'\n')
   652	
   653	        try:
   654	            line = await asyncio.wait_for(self._exchange(encoded), timeout=worker_timeout_s)
   655	        except asyncio.TimeoutError as exc:
   656	            self.broken = True
   657	            raise HotSpotTimeoutError(f"HotSpot worker exceeded {worker_timeout_s:g} seconds") from exc
   658	        except asyncio.CancelledError:
   659	            self.broken = True
   660	            raise
   661	        except Exception as exc:
   662	            self.broken = True
   663	            detail = await self._stderr_text()
   664	            raise HotSpotOracleError(detail or "HotSpot worker communication failed") from exc
   665	
   666	        if not line:
   667	            self.broken = True
   668	            try:
   669	                await asyncio.wait_for(self.process.wait(), timeout=1.0)
   670	            except asyncio.TimeoutError:
   671	                pass
   672	            detail = await self._stderr_text()
   673	            raise HotSpotOracleError(detail or "HotSpot worker exited without a response")
   674	        try:
   675	            envelope = json.loads(line.decode("utf-8"))
   676	        except (UnicodeDecodeError, json.JSONDecodeError, ValueError) as exc:
   677	            self.broken = True
   678	            raise HotSpotOracleError("HotSpot worker returned invalid JSON") from exc
   679	        if not isinstance(envelope, dict) or not isinstance(envelope.get("ok"), bool):
   680	            self.broken = True
   681	            raise HotSpotOracleError("HotSpot worker returned an invalid response envelope")
   682	        if envelope["ok"] is not True:
   683	            self.broken = True
   684	            error = envelope.get("error")
   685	            detail = error[:2000] if isinstance(error, str) else "HotSpot evaluation failed"
   686	            if envelope.get("error_type") == "TimeoutError":
   687	                raise HotSpotTimeoutError(detail or "HotSpot exceeded its solver timeout")
   688	            raise HotSpotOracleError(detail)
   689	        response = envelope.get("result")
   690	        if not isinstance(response, dict):
   691	            self.broken = True
   692	            raise HotSpotOracleError("HotSpot worker returned an invalid result")
   693	        return response
   694	
   695	    async def close(self) -> None:
   696	        self.broken = True
   697	        started = time.perf_counter()
   698	        try:
   699	            await _terminate_process_group(self.process, immediate=getattr(self, 'immediate_kill', False))
   700	        finally:
   701	            self.kill_latency_s = time.perf_counter() - started
   702	            stdout_task = getattr(self, '_stdout_task', None)
   703	            if stdout_task is not None:
   704	                try:
   705	                    await asyncio.wait_for(asyncio.shield(stdout_task), timeout=1.0)
   706	                except asyncio.TimeoutError:
   707	                    stdout_task.cancel()
   708	                    await asyncio.gather(stdout_task, return_exceptions=True)
   709	                    self._stdout_status['truncated'] = True
   710	                    self._write_stdout_status()
   711	            if not self._stderr_task.done():
   712	                try:
   713	                    await asyncio.wait_for(asyncio.shield(self._stderr_task), timeout=1.0)
   714	                except asyncio.TimeoutError:
   715	                    self._stderr_task.cancel()
   716	                    await asyncio.gather(self._stderr_task, return_exceptions=True)
   717	            if getattr(self, '_retained_directory', None) is not None:
   718	                destination = self._retained_directory / 'worker-stderr.bin'
   719	                if self._stderr_file == destination:
   720	                    pass
   721	                elif self._stderr_file.exists():
   722	                    shutil.copyfile(self._stderr_file, destination)
   723	                else:
   724	                    destination.touch()
   725	            self.workspace.cleanup()

   728	class _WarmWorkerPool:
   729	    """Run-scoped pool: a worker is leased by one evaluation at a time."""
   730	
   731	    def __init__(self, max_workers: int):
   732	        self.loop = asyncio.get_running_loop()
   733	        self._max_workers = max_workers
   734	        self._capacity = asyncio.Semaphore(max_workers)
   735	        self._available: asyncio.Queue[_WarmWorker] = asyncio.Queue()
   736	        self._workers: set[_WarmWorker] = set()
   737	        self._condition = asyncio.Condition()
   738	        self._active = 0
   739	        self._closing = False
   740	        self._closed = False
   741	
   742	    async def _retire(self, worker: _WarmWorker) -> None:
   743	        self._workers.discard(worker)
   744	        await _uncancellable_cleanup(worker.close())
   745	
   746	    async def _spawn(self, *, immediate_kill=False) -> _WarmWorker:
   747	        task = asyncio.create_task(_WarmWorker.start())
   748	        try:
   749	            worker = await asyncio.shield(task)
   750	        except asyncio.CancelledError as exc:
   751	            worker = await task
   752	            worker.immediate_kill = immediate_kill
   753	            await _uncancellable_cleanup(worker.close())
   754	            exc.kill_latency_s = worker.kill_latency_s
   755	            raise
   756	        worker.immediate_kill = immediate_kill
   757	        self._workers.add(worker)
   758	        return worker
   759	
   760	    @asynccontextmanager
   761	    async def lease(self, *, immediate_kill=False):
   762	        if asyncio.get_running_loop() is not self.loop:
   763	            raise RuntimeError("HotSpotOracle workers must be closed before using the oracle on another event loop")
   764	        await self._capacity.acquire()
   765	        active = False
   766	        worker = None
   767	        try:
   768	            async with self._condition:
   769	                if self._closing:
   770	                    raise RuntimeError("HotSpotOracle worker pool is closed")
   771	                self._active += 1
   772	                active = True
   773	            while worker is None:
   774	                try:
   775	                    candidate = self._available.get_nowait()
   776	                except asyncio.QueueEmpty:
   777	                    worker = await self._spawn(immediate_kill=immediate_kill)
   778	                else:
   779	                    if candidate.healthy:
   780	                        worker = candidate
   781	                    else:
   782	                        await _uncancellable_cleanup(self._retire(candidate))
   783	            try:
   784	                yield worker
   785	            except BaseException:
   786	                retiring, worker = worker, None
   787	                await _uncancellable_cleanup(self._retire(retiring))
   788	                raise
   789	            else:
   790	                if worker.healthy and not self._closing:
   791	                    self._available.put_nowait(worker)
   792	                else:
   793	                    retiring, worker = worker, None
   794	                    await _uncancellable_cleanup(self._retire(retiring))
   795	                if worker is not None:
   796	                    worker = None
   797	        finally:
   798	            if worker is not None:
   799	                retiring, worker = worker, None
   800	                await _uncancellable_cleanup(self._retire(retiring))
   801	            if active:
   802	                async with self._condition:
   803	                    self._active -= 1
   804	                    self._condition.notify_all()
   805	            self._capacity.release()
   806	
   807	    async def close(self) -> None:
   808	        if self._closed:
   809	            return
   810	        if asyncio.get_running_loop() is not self.loop:
   811	            raise RuntimeError("HotSpotOracle worker pool must close on its owning event loop")
   812	        async with self._condition:
   813	            self._closing = True
   814	            await self._condition.wait_for(lambda: self._active == 0)
   815	        workers = tuple(self._workers)
   816	        self._workers.clear()
   817	        await asyncio.gather(*(_uncancellable_cleanup(worker.close()) for worker in workers))
   818	        self._closed = True

   938	    async def __call__(self, candidate: Design, *, timeout_s: float | None = None) -> HotSpotThermalResult:
   939	        if timeout_s is not None and (not math.isfinite(timeout_s) or timeout_s <= 0):
   940	            raise ValueError('call timeout must be positive and finite')
   941	        self.last_kill_latency_s = None
   942	        evaluation_started = time.perf_counter()
   943	        # All legality checks happen before the worker is created or HotSpot is run.
   944	        centers = self.centers_for(candidate)
   945	        request = {
   946	            "candidate_hash": candidate.hash,
   947	            "problem_hash": self.problem_hash,
   948	            "centers_mm": centers,
   949	            # The worker turns these dies (TAP's own rotate) before it writes the floorplan.
   950	            "rotated": list(self.geometry.rotated(candidate)),
   951	            "grid_rows": self.grid_rows,
   952	            "grid_cols": self.grid_cols,
   953	            "solver_timeout_s": self.solver_timeout_s,
   954	            "expected_binary_sha256": self.problem.hotspot_binary_sha256,
   955	            "hotspot_binary": str(self.hotspot_binary),
   956	            "expected_source_sha256": dict(self.problem.generator_sha256),
   957	            "tap_parameters": {
   958	                "interposer_size_mm": self.problem.interposer_size_mm,
   959	                "widths_mm": self.problem.widths_mm,
   960	                "heights_mm": self.problem.heights_mm,
   961	                "powers_w": self.problem.powers_w,
   962	                "connections": self.problem.connections,
   963	                "link_type": self.problem.link_type,
   964	                "granularity_mm": self.problem.granularity_mm,
   965	                "bump_halos_mm": self.problem.bump_halos_mm,
   966	                "ambient_c": self.problem.ambient_c,
   967	            },
   968	        }
   969	        if self.retained_output_root is not None:
   970	            request['retained_output_root'] = str(self.retained_output_root)
   971	            request['image_identity'] = getattr(self, 'image_identity', {'status': 'unavailable'})
   972	        if timeout_s is not None:
   973	            request['solver_timeout_s'] = min(self.solver_timeout_s, timeout_s)
   974	            request['bounded_call'] = True
   975	        try:
   976	            response = await _run_worker(
   977	                request,
   978	                worker_timeout_s=self.worker_timeout_s if timeout_s is None else min(self.worker_timeout_s, timeout_s),
   979	                pool=self._get_worker_pool(),
   980	                validator=lambda value: self._result_from_response(value, candidate, centers, evaluation_started),
   981	            )
   982	        except BaseException as exc:
   983	            self.last_kill_latency_s = getattr(exc, 'kill_latency_s', None)
   984	            if timeout_s is not None and isinstance(exc, (asyncio.CancelledError, HotSpotTimeoutError)):
   985	                self._retain_interrupted_output(timeout_s)
   986	            raise
   987	        # Tests and adapters may replace _run_worker with a simple response stub;
   988	        # the production pool validates inside the lease so malformed replies retire it.
   989	        if isinstance(response, HotSpotThermalResult):
   990	            return response
   991	        return self._result_from_response(response, candidate, centers, evaluation_started)

   993	    def _retain_interrupted_output(self, timeout_s):
   994	        if self.retained_output_root is None:
   995	            return
   996	        for output in self.retained_output_root.glob('evaluation-*'):
   997	            for name in ('stdout', 'stderr'):
   998	                binary = output / f'{name}.bin'
   999	                if binary.exists():
  1000	                    (output / f'{name}.txt').write_text(binary.read_bytes().decode('utf-8', errors='replace'))
  1001	            if not (output / 'EXIT.json').exists():
  1002	                (output / 'EXIT.json').write_text(json.dumps({
  1003	                    'status': 'parent_timeout', 'returncode': None, 'timeout_s': timeout_s,
  1004	                    'kill_latency_s': self.last_kill_latency_s}) + '\n')

  1006	    def _result_from_response(self, response: dict, candidate: Design,
  1007	                              centers: tuple[tuple[float, float], ...], evaluation_started: float
  1008	                              ) -> HotSpotThermalResult:
  1009	        if response.get("candidate_hash") != candidate.hash or response.get("problem_hash") != self.problem_hash:
  1010	            raise HotSpotOracleError("worker result is not bound to the requested design and problem")
  1011	        if response.get("binary_sha256") != self.problem.hotspot_binary_sha256:
  1012	            raise HotSpotOracleError("worker HotSpot binary differs from the problem contract")
  1013	        if response.get("grid_rows") != self.grid_rows or response.get("grid_cols") != self.grid_cols:
  1014	            raise HotSpotOracleError("worker returned a different grid resolution")
  1015	
  1016	        rows = response.get("heatmap_c")
  1017	        if (not isinstance(rows, list) or len(rows) != self.grid_rows or
  1018	                any(not isinstance(row, list) or len(row) != self.grid_cols for row in rows)):
  1019	            raise HotSpotOracleError("worker returned an incomplete chip-layer heatmap")
  1020	        heatmap = tuple(tuple(_number(value, "heatmap temperature") for value in row) for row in rows)
  1021	        peak = _number(response.get("maximum_temperature_c"), "peak temperature")
  1022	        if not math.isclose(peak, max(max(row) for row in heatmap), rel_tol=0, abs_tol=1e-9):
  1023	            raise HotSpotOracleError("worker peak does not match its chip-layer heatmap")
  1024	        all_layers_peak = _number(response.get("all_layers_peak_c"), "all-layer peak temperature")
  1025	        if all_layers_peak < peak:
  1026	            raise HotSpotOracleError("worker all-layer peak is below its chip-layer peak")
  1027	        solver_seconds = _number(response.get("solver_seconds"), "solver runtime")
  1028	        generation_seconds = _number(response.get("input_generation_seconds"), "input generation runtime")
  1029	        if solver_seconds < 0 or generation_seconds < 0:
  1030	            raise HotSpotOracleError("worker returned a negative runtime")
  1031	        input_hashes = response.get("input_sha256")
  1032	        required_inputs = {
  1033	            "new_hotspot.config", "sampleL4_ChipLayer.flp", "sample.ptrace", "samplelayers.lcf",
  1034	            "sampleL0_Substrate.flp", "sampleL1_C4Layer.flp", "sampleL2_Interposer.flp",
  1035	            "sampleL3_UbumpLayer.flp", "sampleL5_TIM.flp",
  1036	        }
  1037	        if (not isinstance(input_hashes, dict) or not required_inputs.issubset(input_hashes) or
  1038	                any(not isinstance(name, str) or not isinstance(digest, str) or len(digest) != 64
  1039	                    or any(character not in "0123456789abcdef" for character in digest)
  1040	                    for name, digest in input_hashes.items())):
  1041	            raise HotSpotOracleError("worker returned invalid generated-input hashes")
  1042	
  1043	        return HotSpotThermalResult(
  1044	            candidate.hash,
  1045	            peak,
  1046	            self.solver_version,
  1047	            converged=True,
  1048	            cost_microdollars=0,
  1049	            evidence_kind=f"HotSpot 7 steady-state TAP {self.problem.fixture_id} grid; Manhattan wire proxy, not CPLEX routing",
  1050	            fixture_id=self.problem.fixture_id,
  1051	            config_sha256=self.problem.config_sha256,
  1052	            phase_timings=response.get('phase_timings', {}),
  1053	            problem_hash=self.problem_hash,
  1054	            wirelength_mm=self._wirelength_mm(centers),
  1055	            wire_metric="connectivity-weighted Manhattan center distance (mm); not routed CPLEX length",
  1056	            heatmap_c=heatmap,
  1057	            grid_rows=self.grid_rows,
  1058	            grid_cols=self.grid_cols,
  1059	            ambient_c=self.problem.ambient_c,
  1060	            solver_seconds=solver_seconds,
  1061	            evaluation_seconds=time.perf_counter() - evaluation_started,
  1062	            input_generation_seconds=generation_seconds,
  1063	            input_sha256=tuple(sorted(input_hashes.items())),
  1064	            all_layers_peak_c=all_layers_peak,
  1065	        )

  1068	async def _run_worker(request: dict, *, worker_timeout_s: float,
  1069	                      pool: _WarmWorkerPool | None = None, validator=None):
  1070	    """Send one request to a leased warm worker; standalone calls own a short-lived pool."""
  1071	    owns_pool = pool is None
  1072	    pool = pool or _WarmWorkerPool(max_workers=1)
  1073	    worker = None
  1074	    try:
  1075	        async with pool.lease(immediate_kill=request.get('bounded_call', False)) as worker:
  1076	            response = await worker.evaluate(request, worker_timeout_s=worker_timeout_s)
  1077	            return validator(response) if validator is not None else response
  1078	    except BaseException as exc:
  1079	        exc.kill_latency_s = getattr(worker, 'kill_latency_s', getattr(exc, 'kill_latency_s', None))
  1080	        raise
  1081	    finally:
  1082	        if owns_pool:
  1083	            await _uncancellable_cleanup(pool.close())
