   209	def displacement(raw_centers, centers) -> dict:
   210	    """Raw-to-layout displacement under two named norms: per coordinate (L-infinity) and Euclidean per die."""
   211	    per_die = []
   212	    for die, (r, c) in enumerate(zip(raw_centers, centers)):
   213	        dx, dy = c[0] - r[0], c[1] - r[1]
   214	        per_die.append({"die": die, "dx_mm": _mm(dx), "dy_mm": _mm(dy), "euclidean_mm": round(math.hypot(dx, dy), 6)})
   215	    return {"per_die": per_die,
   216	            "max_per_coordinate_linf_mm": _mm(max(max(abs(d["dx_mm"]), abs(d["dy_mm"])) for d in per_die)),
   217	            "max_euclidean_mm": max(d["euclidean_mm"] for d in per_die)}

   221	    """Refuse unless SciPy is the version requirements.txt pins; return the checked versions for the record.
   222	
   223	    Only SciPy is pinned. The HiGHS version is whatever this SciPy build reports: it is recorded, not pinned,

   228	    if pin is None or versions["scipy"] != pin.group(1):

   253	def minimal_snaps(geometry: TapGeometry, raw_centers, rotated, grid_mm: float) -> tuple[tuple, dict]:
   254	    """Every legal layout reachable from the rounded raw centres with the fewest grid-step moves.
   255	
   256	    Conflict-directed: from the current layout it branches only on one-step moves of the dies in the
   257	    first remaining violation. That is complete, because no move of a third die can clear a margin or
   258	    pair violation. Returns the rounded centres and {steps per die: centres}.
   259	    """
   260	    count = len(raw_centers)
   261	    rounded = tuple(tuple(round_to_lattice(value, grid_mm) for value in point) for point in raw_centers)
   262	
   263	    def centers_for(steps):
   264	        return tuple((rounded[i][0] + steps[i][0] * grid_mm, rounded[i][1] + steps[i][1] * grid_mm)
   265	                     for i in range(count))
   266	
   267	    solutions = {}
   268	    for depth in range(MAX_SNAP_STEPS + 1):
   269	        steps = [[0, 0] for _ in range(count)]
   270	
   271	        def search(remaining):
   272	            problems = _violations(geometry, centers_for(steps), rotated)
   273	            if not problems:
   274	                if remaining == 0:
   275	                    key = tuple(tuple(step) for step in steps)
   276	                    solutions[key] = centers_for(steps)
   277	                return
   278	            if remaining == 0:
   279	                return
   280	            for die in problems[0].dies:
   281	                for axis in (0, 1):
   282	                    for sign in (1, -1):
   283	                        if steps[die][axis] * sign < 0:  # never undo a step: a minimal set has no pair
   284	                            continue
   285	                        steps[die][axis] += sign
   286	                        search(remaining - 1)
   287	                        steps[die][axis] -= sign
   288	
   289	        search(depth)
   290	        if solutions:
   291	            return rounded, solutions
   292	    raise ValueError(f"no legal {grid_mm:g} mm layout within {MAX_SNAP_STEPS} grid steps of the raw centres")

   295	def snap_to_grid(geometry: TapGeometry, raw_centers, rotated, *, decode_precision_mm: float,
   296	                 grid_mm: float = SNAP_GRID_MM) -> dict:
   297	    """Round each centre to the nearest grid node, then make the fewest grid-step moves that make
   298	    the layout legal on the fixture (minimal_snaps). Among the move sets of minimal size it keeps the
   299	    one closest to the raw centres (sum of squared distances), then the lexicographically smallest
   300	    centres. If the runner-up is closer than the decode precision can separate, the result says so.
   301	    """
   302	    rounded, solutions = minimal_snaps(geometry, raw_centers, rotated, grid_mm)
   303	    count = len(raw_centers)
   304	
   305	    def centers_for(steps):
   306	        return tuple((rounded[i][0] + steps[i][0] * grid_mm, rounded[i][1] + steps[i][1] * grid_mm)
   307	                     for i in range(count))
   308	
   309	    def cost(centers):
   310	        return sum((c[0] - r[0]) ** 2 + (c[1] - r[1]) ** 2 for c, r in zip(centers, raw_centers))
   311	
   312	    ranked = sorted(solutions.items(), key=lambda item: (cost(item[1]), item[1]))
   313	    best_steps, best = ranked[0]
   314	    if not legal_on(geometry, best, rotated)[0]:
   315	        raise AssertionError("snapped layout failed the fixture gate")
   316	    adjustments = []
   317	    for die, (dx, dy) in enumerate(best_steps):
   318	        if dx == 0 and dy == 0:
   319	            continue
   320	        # The reason is exactly what dropping this adjustment would break.
   321	        without = list(best_steps)
   322	        without[die] = (0, 0)
   323	        dropped = _violations(geometry, centers_for(without), rotated)
   324	        if not dropped:
   325	            raise AssertionError(f"adjustment of die {die} is not needed; the move set is not minimal")
   326	        adjustments.append({"die": die, "dx_mm": dx * grid_mm, "dy_mm": dy * grid_mm,
   327	                            "reason": [v.message for v in dropped],
   328	                            "violations_if_dropped": _violation_records(dropped)})
   329	    alternatives = [{"centers_mm": [list(c) for c in centers], "squared_distance_to_raw_mm2": _mm(cost(centers))}
   330	                    for _, centers in ranked[1:4]]
   331	    runner_up = None
   332	    if len(ranked) > 1:
   333	        second = ranked[1][1]
   334	        margin = cost(second) - cost(best)
   335	        # Moving each raw coordinate by up to the decode precision shifts the cost difference by at most this.
   336	        bound = sum(2 * decode_precision_mm * (abs(b - r) + abs(s - r))
   337	                    for bp, sp, rp in zip(best, second, raw_centers) for b, s, r in zip(bp, sp, rp) if b != s)
   338	        runner_up = {"squared_distance_margin_mm2": _mm(margin), "decode_precision_bound_mm2": _mm(bound),
   339	                     "decided_below_decode_precision": margin <= bound}
   340	    return {
   341	        "rounded_centers_mm": [list(c) for c in rounded],
   342	        "rounded_violations": _violation_records(_violations(geometry, rounded, rotated)),
   343	        "grid_steps": int(sum(abs(s) for pair in best_steps for s in pair)),
   344	        "adjustments": adjustments,
   345	        "centers_mm": [list(c) for c in best],
   346	        "squared_distance_to_raw_mm2": _mm(cost(best)),
   347	        "alternatives_same_step_count": alternatives,
   348	        "runner_up": runner_up,
   349	    }

   352	def _precision_evidence(decode: dict) -> dict:
   353	    """TAP's own placer puts centres on the 1 mm grid, so its panels measure the decode error."""
   354	    deviations = []
   355	    for figure in decode["figures"].values():
   356	        for panel_name, panel in figure["panels"].items():
   357	            if panel_name == "a":
   358	                continue
   359	            for chiplet in panel["chiplets"]:
   360	                deviations.extend(abs(v - round(v)) for v in chiplet["decoded_center_mm"])
   361	    return {"tap_placer_panels": "fig5 b,c,d and fig6 b,c (TAP output, 1 mm grid by construction)",
   362	            "max_center_offset_from_grid_mm": _mm(max(deviations)), "centers_checked": len(deviations) // 2,
   363	            "norm": "per coordinate (L-infinity)",
   364	            "kind": ("empirical estimate under stated assumptions: the TAP-placer panels sit exactly on its 1 mm grid, "
   365	                     "and the drawing maps every panel the same way. It is not a figure-to-silicon error bound."),
   366	            "reuse": ("These five panels are also the router's validation points, so the two checks are not "
   367	                      "independent of each other.")}

   370	def _routed_mm(problem, centers, rotated, link_type) -> tuple[float, dict]:
   371	    """The builder's only route. The SciPy pin is checked before the optimizer runs, so a mismatched stack is
   372	    refused with zero optimizer calls.
   373	
   374	    Returns the routed total, rounded to 0.1 mm (every committed value sits at least 0.015 mm from a rounding
   375	    boundary), and the solver versions checked for this route, for the record.
   376	    """
   377	    solver = pinned_solver()
   378	    wire = routed_wirelength(problem, centers, rotated, link_type=link_type)
   379	    return round(wire.total_mm, 1), solver

   387	    """Per-coordinate decode uncertainty (mm), rounded up: TAP-grid decode error, plus the coordinate times

   389	    scales = panel["scale_pt_per_mm"]
   390	    estimates = (scales["x"], scales["y"], scales["from_die_outlines"])
   391	    spread = (max(estimates) - min(estimates)) / min(estimates)
   392	    return [math.ceil((grid_error_mm + abs(v) * spread + 0.00005) * 1e4) / 1e4 for v in center]

   681	def _envelope(bases: list[dict], timing: dict) -> dict:
   682	    """Calls for each base layout: itself, plus each die +/-0.5 mm in x and separately in y."""
   683	    reserve = STARTUP_RESERVE_S + timing["reap_s"] + ARCHIVE_RESERVE_S
   684	    plans, forecast, reserved, total = [], 0.0, 0.0, 0
   685	    for base in bases:
   686	        geometry = TapGeometry.load(base["fixture_id"])
   687	        rotated = tuple(base["rotated"])
   688	        start = [list(c) for c in base["centers_mm"]]
   689	        calls, excluded = [], []
   690	        options = [("base", None, None, 0.0)] + [
   691	            (f"die{die}/{axis}{'+' if delta > 0 else '-'}{PERTURBATION_MM:g}", die, axis, delta)
   692	            for die in range(len(start)) for axis in ("x", "y") for delta in (PERTURBATION_MM, -PERTURBATION_MM)]
   693	        for name, die, axis, delta in options:
   694	            centers = [list(c) for c in start]
   695	            if die is not None:
   696	                centers[die][0 if axis == "x" else 1] += delta
   697	            entry = {"call_id": f"{base['id']}/{name}", "die": die, "axis": axis, "delta_mm": delta}
   698	            legal, reason = legal_on(geometry, centers, rotated)
   699	            if not legal:
   700	                excluded.append({**entry, "fixture_gate_message": reason,
   701	                                 "violations": _violation_records(_violations(geometry, centers, rotated))})
   702	                continue
   703	            design = geometry.design_from_centers(centers, rotated)
   704	            calls.append({**entry, "centers_mm": centers, "rotated": list(rotated), "design_hash": design.hash,
   705	                          "wire_proxy_mm": _mm(geometry.wirelength_mm(geometry.centers_for(design)))})
   706	        if calls[0]["call_id"] != f"{base['id']}/base":
   707	            raise ValueError("the unperturbed base layout must be legal")
   708	        family = base["fixture_id"].removesuffix("-45mm")
   709	        median = timing["median_64_s"][family]
   710	        forecast += len(calls) * (median + timing["overhead_s"])
   711	        reserved += len(calls) * (median + reserve)
   712	        total += len(calls)
   713	        plans.append({**{k: v for k, v in base.items() if k not in ("centers_mm", "rotated")},
   714	                      "timing_family": family, "grid": ENVELOPE_GRID,
   715	                      "considered": len(options), "expected_call_count": len(calls),
   716	                      "excluded_count": len(excluded), "calls": calls, "excluded": excluded})
   717	    worst_call = timing["worst_64_s"] + reserve
   718	    failed = math.ceil(FAILED_ATTEMPT_FRACTION * total) * worst_call
   719	    cap = reserved + failed + worst_call + timing["tail_s"]
   720	    return {"plans": plans, "total": total, "forecast": forecast, "reserved": reserved, "failed": failed,
   721	            "worst_call": worst_call, "cap": cap}

   725	    timing = _timing_inputs()

   818	def build_all() -> dict[Path, bytes]:
   819	    decode = json.loads(DECODE.read_text(encoding="utf-8"))
   820	    references = [build_reference(spec, decode) for spec in REFERENCES]
   821	    outputs = {REFERENCES_DIR / f"{ref['reference_id']}.json": _json_bytes(ref) for ref in references}
   822	    variants = [build_variant(ref, decode) for ref in references]
   823	    outputs.update({REFERENCES_DIR / f"{v['variant_id']}.json": _json_bytes(v) for v in variants})
   824	    outputs[ENVELOPE_PLAN] = _json_bytes(build_envelope_plan(references, variants))
   825	    return outputs

   828	def main(argv=None) -> int:
   829	    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
   830	    mode = parser.add_mutually_exclusive_group(required=True)
   831	    mode.add_argument("--write", action="store_true")
   832	    mode.add_argument("--check", action="store_true")
   833	    args = parser.parse_args(argv)
   834	    outputs = build_all()
   835	    if args.write:
   836	        for path, data in outputs.items():
   837	            path.write_bytes(data)
   838	            print(f"wrote {_rel(path)} {hashlib.sha256(data).hexdigest()}")
   839	        return 0
   840	    stale = [_rel(path) for path, data in outputs.items() if not path.is_file() or path.read_bytes() != data]
   841	    for name in stale:
   842	        print(f"stale: {name}", file=sys.stderr)
   843	    return 1 if stale else 0
